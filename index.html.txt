        let usedIndices = [];
        let hasShownTodayFortune = false;
        let showingFortune = false;
        let allMessagesShown = false;

        function getTodayString() {
            const d = new Date();
            return d.getFullYear() + '-' + String(d.getMonth() + 1).padStart(2, '0') + '-' + String(d.getDate()).padStart(2, '0');
        }

        function loadData() {
            const today = getTodayString();
            const savedDate = localStorage.getItem('lastShownDate');
            
            if (savedDate === today) {
                hasShownTodayFortune = true;
                const savedFortune = localStorage.getItem('todayFortune');
                const savedTitle = localStorage.getItem('todayTitle');
                if (savedFortune && savedTitle) {
                    document.getElementById('fortuneText').textContent = savedFortune;
                    document.getElementById('fortuneTitle').textContent = savedTitle;
                }
            }

            const saved = localStorage.getItem('usedIndices');
            if (saved) {
                try {
                    usedIndices = JSON.parse(saved);
                    if (usedIndices.length >= fortunes.length) {
                        allMessagesShown = true;
                    }
                } catch(e) {
                    usedIndices = [];
                }
            }
        }

        function generateFortune() {
            if (hasShownTodayFortune) {
                showFortune();
                return;
            }

            if (usedIndices.length >= fortunes.length) {
                allMessagesShown = true;
                showAllMessagesView();
                return;
            }

            const available = [];
            for (let i = 0; i < fortunes.length; i++) {
                if (usedIndices.indexOf(i) === -1) {
                    available.push(i);
                }
            }

            const index = available[Math.floor(Math.random() * available.length)];
            usedIndices.push(index);
            localStorage.setItem('usedIndices', JSON.stringify(usedIndices));

            const fortune = fortunes[index];
            const title = titles[Math.floor(Math.random() * titles.length)];

            const today = getTodayString();
            localStorage.setItem('lastShownDate', today);
            localStorage.setItem('todayFortune', fortune);
            localStorage.setItem('todayTitle', title);

            hasShownTodayFortune = true;
            document.getElementById('fortuneText').textContent = fortune;
            document.getElementById('fortuneTitle').textContent = title;

            showFortune();
        }

        function showFortune() {
            showingFortune = true;
            document.getElementById('constellationView').classList.add('hidden');
            document.getElementById('fortuneCard').classList.remove('hidden');
            setTimeout(function() {
                document.getElementById('fortuneCard').classList.add('show');
            }, 50);
            document.getElementById('mainButton').textContent = "Remember that you are wonderful!";
        }

        function returnToStart() {
            showingFortune = false;
            document.getElementById('fortuneCard').classList.remove('show');
            setTimeout(function() {
                document.getElementById('fortuneCard').classList.add('hidden');
                document.getElementById('constellationView').classList.remove('hidden');
            }, 600);
            document.getElementById('mainButton').textContent = "To the closest person";
        }

        function showAllMessagesView() {
            document.getElementById('constellationView').classList.add('hidden');
            document.getElementById('fortuneCard').classList.add('hidden');
            document.getElementById('allMessagesView').classList.remove('hidden');
            document.getElementById('mainButton').textContent = "Thank you for being you ❤️";
        }

        function handleButtonClick() {
            if (allMessagesShown) return;
            
            if (showingFortune) {
                returnToStart();
            } else {
                generateFortune();
            }
        }

        window.onload = function() {
            loadData();
            
            setTimeout(function() {
                document.getElementById('welcomeScreen').classList.remove('show');
            }, 2500);

            if (allMessagesShown) {
                setTimeout(function() {
                    showAllMessagesView();
                }, 4000);
            }
        };
